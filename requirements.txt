# Runtime dependencies for training + evaluation, pinned to versions verified
# working locally (macOS, Python 3.9-3.12). On Linux, `pip install torch` pulls
# the CUDA build by default — correct for RunPod RTX 4090 pods.
#
# NOT included (only needed for dSprites/3dshapes dataset *generation*, not
# training): tensorflow, tensorflow-datasets. scripts/setup_pod.sh installs
# tensorflow-cpu on demand if it has to regenerate dSprites.

# PINNED TO LIGHTNING 1.x DELIBERATELY — DO NOT BUMP TO 2.x.
# gc_cbm.py / gc_cem.py use `training_step(..., optimizer_idx=...)`, the
# Lightning 1.x automatic multi-optimizer API that 2.0 removed. Under
# Lightning 2.x the adversarial GC models crash with "Training with multiple
# optimizers is only supported with manual optimization". Moving to 2.x needs
# a real refactor of the adversarial step, not a version bump.
# torch is capped at 2.3.1 per the original setup.py; torchvision 0.18.1 is
# its matching release. All have cp312 wheels.
torch==2.3.1
torchvision==0.18.1
pytorch-lightning==1.9.5
torchmetrics==0.11.4

numpy==1.26.0
# Lightning 1.9.5 calls pkg_resources.declare_namespace, removed in
# setuptools 70. Python 3.12 also no longer bundles setuptools at all.
setuptools<70.0.0
scipy==1.13.1
scikit-learn==1.5.1
pandas==2.2.2
joblib==1.4.2
h5py==3.11.0
Pillow==10.4.0

matplotlib==3.9.2
seaborn==0.13.2

PyYAML==6.0.1
mergedeep==1.3.4
prettytable==3.17.0
tqdm==4.66.5

# W&B is used on HPC/RunPod but is not installed in the local dev env, so this
# is the one dependency not pinned from a verified version. Any 0.18+ release
# works with the wandb.init/log/finish + WandbLogger APIs this codebase uses.
wandb>=0.18,<1.0
